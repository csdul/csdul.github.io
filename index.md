<!-- ABOUT -->
<section class="section">
  <div class="container narrow">
    <div class="section-label">ABOUT CSDUL</div>

    <h2>A research infrastructure for studying the social determinants of health</h2>

    <p class="lead">
      The Canadian Social Determinants Urban Laboratory (CSDUL) brings
      together Canadian administrative, survey, census, environmental,
      socioeconomic, and health-system data within a common and
      reproducible research framework.
    </p>

    <p class="lead">
      CSDUL organizes these data through a central Individual Health Hub
      and five thematic Nodes representing major dimensions of the social
      determinants of health.
    </p>
  </div>
</section>
